Nous avons besoin de fonctions pures pour calculer la distance euclidienne, l'énergie consommée et le temps de vol d'un point A à un point B en fonction de la charge embarquée.

def distance(p1, p2):
    return math.hypot(p2[0] - p1[0], p2[1] - p1[1])

def calculer_energie(dist_m, charge_kg):
    energie_base = CONSO_BASE_WH_M * dist_m
    energie_extra = (PENALITE_CONSO_WH_M * charge_kg) * dist_m
    return energie_base + energie_extra

def calculer_temps(dist_m, charge_kg):
    vitesse_reelle = VITESSE_BASE_MS * (1 - (PENALITE_VITESSE * charge_kg))
    return dist_m / vitesse_reelle

Une fonction dédiée qui vérifie si un segment de vol croise l'intervalle de pluie. Si oui, elle ajoute la durée de l'averse au temps total, simulant ainsi la "pause" en plein vol.

def appliquer_meteo(temps_debut, temps_fin):
    if temps_debut < N1 and temps_fin > N0:
        duree_pluie = min(temps_fin, N1) - max(temps_debut, N0)

Étape 4 : L'Algorithme de Routage (Cœur du système)
Pour faire simple et robuste en moins de 4 heures, nous implémentons une stratégie gloutonne (Greedy) : le drone part de la Citadelle, va à la cible non livrée la plus proche, dépose le colis, puis rentre à vide pour recharger.

def executer_missions():
    temps_global = T0
    energie_totale = 0
    trajectoires = [] 

    while any(c["reste"] > 0 for c in cibles.values()):
        cible_proche = None
        dist_min = float('inf')

        # Trouver la cible la plus proche qui nécessite encore une livraison
        for nom, data in cibles.items():
            if data["reste"] > 0:
                d = distance(CITADELLE, data["pos"])
                if d < dist_min:
                    dist_min = d
                    cible_proche = nom

        data_cible = cibles[cible_proche]
        
        # Trajet Aller (chargé)
        t_aller = calculer_temps(dist_min, data_cible["masse"])
        pause_aller = appliquer_meteo(temps_global, temps_global + t_aller)
        temps_global += t_aller + pause_aller
        e_aller = calculer_energie(dist_min, data_cible["masse"])
        
        # Mise à jour
        data_cible["reste"] -= 1
        trajectoires.append((CITADELLE, data_cible["pos"], temps_global))

        # Trajet Retour (à vide)
        t_retour = calculer_temps(dist_min, 0)
        pause_retour = appliquer_meteo(temps_global, temps_global + t_retour)
        temps_global += t_retour + pause_retour
        e_retour = calculer_energie(dist_min, 0)

        trajectoires.append((data_cible["pos"], CITADELLE, temps_global))
        energie_totale += (e_aller + e_retour)

    return trajectoires, temps_global, energie_totale

vols, temps_final, conso_totale = executer_missions()

def distance(p1, p2):

def calc_temps(dist, masse):

import math
import matplotlib.pyplot as plt
from donnees import *

def distance(p1, p2):
    # Application du théorème de Pythagore pour trouver la ligne droite
    # On soustrait les X et les Y, on les met au carré, et on prend la racine
    return math.hypot(p2[0] - p1[0], p2[1] - p1[1])

def calc_energie(dist, masse):
    # L'énergie totale est l'énergie de base PLUS la pénalité de poids
    # On multiplie la pénalité par la masse embarquée, on ajoute à la base, puis on multiplie par les mètres
    return (CONSO_BASE + (PENALITE_E * masse)) * dist

def calc_temps(dist, masse):
    # Le poids freine le drone. On calcule le pourcentage de vitesse restante (ex: 1 - 0.15 * 0.6 = 91%)
    vitesse = VITESSE_BASE * (1 - (PENALITE_V * masse))
    # Le temps en secondes est simplement la distance divisée par cette vitesse réelle
    return dist / vitesse

def verif_meteo(t_deb, t_fin):
    # On vérifie si l'heure de vol (t_deb à t_fin) croise l'heure de la pluie (N0 à N1)
    if t_deb < N1 and t_fin > N0:
        # Si oui, on calcule la durée exacte passée sous la pluie pour mettre le drone en pause
        return min(t_fin, N1) - max(t_deb, N0)
    # Si le vol est en dehors des heures de pluie, aucune pause (0 seconde)
    return 0

def executer_planification():
    # Initialisation du chronomètre, du compteur électrique et de la boîte noire (historique)
    t_global = T0
    e_totale = 0
    trajets = []

    # Tant qu'il reste au moins 1 colis à livrer n'importe où
    while any(c["reste"] > 0 for c in CIBLES.values()):
        cible_proche = None
        d_min = float('inf')

        # On scanne toutes les cibles pour trouver la plus proche qui a encore besoin de colis
        for nom, data in CIBLES.items():
            if data["reste"] > 0:
                d = distance(CITADELLE, data["pos"])
                if d < d_min:
                    d_min = d
                    cible_proche = nom

        cible = CIBLES[cible_proche]

        # --- PHASE 1 : LE VOL ALLER (Chargé) ---
        # Calcul du temps avec le poids du colis
        t_aller = calc_temps(d_min, cible["masse"])
        # On vérifie si on s'est pris la pluie sur le chemin
        pause_aller = verif_meteo(t_global, t_global + t_aller)
        # On avance le chronomètre global
        t_global += t_aller + pause_aller
        # On calcule l'énergie consommée avec le poids
        e_aller = calc_energie(d_min, cible["masse"])

        # On valide la livraison et on l'enregistre dans l'historique
        cible["reste"] -= 1
        trajets.append((CITADELLE, cible["pos"], t_global))

        # --- PHASE 2 : LE VOL RETOUR (À vide) ---
        # Même logique, mais la masse est de 0 kg
        t_retour = calc_temps(d_min, 0)
        pause_retour = verif_meteo(t_global, t_global + t_retour)
        t_global += t_retour + pause_retour
        e_retour = calc_energie(d_min, 0)

        # On enregistre le retour à la Citadelle et on cumule l'énergie
        trajets.append((cible["pos"], CITADELLE, t_global))
        e_totale += e_aller + e_retour

    return trajets, t_global, e_totale

PROBLÉMATIQUE PRINCIPALE :
Comment garantir l'efficacité des livraisons par drone dans un 
environnement urbain complexe comme Lille, tout en minimisant la 
consommation énergétique et en s'adaptant aux aléas météorologiques ?

RÉPONSES APPORTÉES PAR LA SOLUTION ALGORITHMIQUE (PYTHON) :

1. GARANTIE DE L'EFFICACITÉ (Optimisation des Trajets)
----------------------------------------------------------------------
- Méthode : Utilisation d'un algorithme "glouton" (Greedy).
- Résultat : Le système sélectionne toujours la cible prioritaire la 
  plus proche. Le calcul de la distance euclidienne stricte empêche 
  tout détour inutile, garantissant le chemin mathématiquement le 
  plus court pour traiter l'environnement urbain dense.

2. MINIMISATION ÉNERGÉTIQUE (Autonomie)
----------------------------------------------------------------------
- Méthode : Comptabilité dynamique de la consommation.
- Résultat : L'algorithme calcule séparément le trajet aller (chargé) 
  et le retour (à vide). L'application d'une pénalité stricte (+10 Wh/km 
  par kg) valide la gestion de la batterie. Le bilan final prouve que 
  l'énergie est consommée au plus juste.

3. RÉALISME OPÉRATIONNEL (Prise en compte des masses)
----------------------------------------------------------------------
- Méthode : Modélisation cinématique liée à la charge (0.5 ou 0.6 kg).
- Résultat : La vitesse nominale (25 km/h) est réduite de 15% pour 
  chaque kilo embarqué. Le chronométrage global est donc parfaitement 
  réaliste et prouve au jury que le planning de livraison est fiable 
  physiquement.

4. ADAPTATION AUX ALÉAS (Jalon Météo)
----------------------------------------------------------------------
- Méthode : Fonction d'interruption temporelle sans déviation spatiale.
- Résultat : Au lieu de provoquer un retour à la base coûteux en 
  énergie, l'algorithme gèle le temps de vol (pause stationnaire) pendant 
  l'averse (entre T=300s et T=600s). Le retard est comptabilisé 
  précisément sans altérer l'optimisation spatiale du trajet.

5. VALIDATION PAR LA PREUVE (Le Livrable Graphique)
----------------------------------------------------------------------
- Méthode : Rendu visuel 2D avec horodatage continu (matplotlib).
- Résultat : La carte générée imprime l'heure d'arrivée en secondes 
  sur chaque point. Le jury peut y lire visuellement les impacts de la 
  masse sur la vitesse et les décalages liés à la pluie, constituant 
  la preuve finale de la résolution du problème.
